use std/fmt.Format
use std/io

struct Point {
    x: i32
    y: i32
}

instance Point {
    impl Format
    method &self.try_format_into(output: &+String): void! {
        output.try_push_str("(")?
        self.x.try_format_into(output)?
        output.try_push_str(", ")?
        self.y.try_format_into(output)?
        output.try_push_str(")")?
        return
    }
}

func main(): i32! {
    let point = Point { x: 3, y: 4 }
    let message = "point = ${point}\n"
    io.print(&message)?
    return 0
}
use std/io
use std/vec.Vec

struct Point {
    x: i32
    y: i32
}

instance Point {
    operator (&self == other: &Self): bool {
        return self.x == other.x && self.y == other.y
    }
}

func main(): i32! {
    let first = Point { x: 3, y: 4 }
    let same = Point { x: 3, y: 4 }
    let other = Point { x: 5, y: 6 }
    if first != same || first == other {
        return 1
    }
    let points = Vec [move first, move other]
    if points[0] != same {
        return 1
    }
    var values = Vec [1, 2, 3]
    values[1] = 4
    if values[0] + values[1] + values[2] != 8 {
        return 1
    }
    io.print("equality found the point\n")?
    return 0
}
struct Buffer {
    values: [i32; 3]
}

instance Buffer {
    operator (&self[index: usize]): &i32 {
        return &self.values[index]
    }

    operator (&+self[index: usize]): &+i32 {
        return &+self.values[index]
    }
}

func at<C>(container: &C, index: usize): i32 where (&C[usize]): &i32 {
    return container[index]
}

func main(): i32 {
    var buffer = Buffer { values: [3, 4, 5] }
    buffer[1] = 9
    if at(&buffer, 1) != 9 {
        return 1
    }
    return 0
}

/// Positive binary32 infinity.
pub const F32_INFINITY: f32 = 1.0f32 / 0.0f32

/// Negative binary32 infinity.
pub const F32_NEG_INFINITY: f32 = -1.0f32 / 0.0f32

/// One quiet binary32 NaN. Arithmetic does not promise to preserve this payload.
pub const F32_NAN: f32 = 0.0f32 / 0.0f32

/// Negative binary64 infinity.

pub const F64_NAN: f64 = 0.0 / 0.0

construct f32 {
    /// Creates a binary32 value with exactly the supplied IEEE representation.
    pub noalloc func from_bits(bits: u32): Self

    /// Rounds a binary64 value to binary32, or returns none if a finite value leaves binary32's
    /// finite nonzero range.
    pub noalloc func from_f64(value: f64): Self?
}

construct f64 {
    /// Creates a binary64 value with exactly the supplied IEEE representation.
    pub noalloc func from_bits(bits: u64): Self
}

instance f32 {
    /// Returns the exact stored IEEE representation.

    pub noalloc method self.is_infinite(): bool

    /// Reports whether this value is finite.
    pub noalloc method self.is_finite(): bool

    /// Reports whether this value is either sign of zero.
    pub noalloc method self.is_zero(): bool

    /// Reports whether this finite nonzero value uses the normal representation.
    pub noalloc method self.is_normal(): bool

    /// Reports whether this value uses the subnormal representation.
    pub noalloc method self.is_subnormal(): bool

    /// Reports whether the IEEE sign bit is set, including for zero and NaN.
    pub noalloc method self.is_sign_negative(): bool

    /// Reports whether the IEEE sign bit is clear, including for zero and NaN.
    pub noalloc method self.is_sign_positive(): bool

    /// Clears the IEEE sign bit without changing the payload.
    pub noalloc method self.abs(): Self

    /// Rounds toward negative infinity.
    pub noalloc method self.floor(): Self

    /// Rounds toward positive infinity.
    pub noalloc method self.ceil(): Self

    /// Rounds toward zero.

    /// Rounds to the nearest integer, choosing the even value at a tie.

    /// Compares every IEEE representation in a deterministic total order.
