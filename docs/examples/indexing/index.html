use std/io.print

func greet(user: User): void! {
    let name = user.display.name() otherwise { "guest" }

    let tone = match user.presence {
        Presence.online { "hello" }
        Presence.away { "welcome back" }
    }

    print("${tone}, ${name}\n")?
}
interface Writer {
    pub method &+self.write(bytes: &[u8]): void!
}

func save_log(output: &+Writer, events: &[Event]): void! {
    for event in events {
        output.write(event.message().bytes())?
        output.write("\n".bytes())?
    }
}
pub struct Session {
    id: SessionId
    user: User
    expires_at: Time
}

pub func Session.start(user: User, clock: &Clock): Session {
    return Session {
        id: SessionId.new(),
        user: move user,
        expires_at: clock.now().plus_minutes(30),
    }
}
func consume(values: Vec<String>): void! {
    for value in move values {
        print(&value as &str)?
    }
}
construct NonEmptyList<T> {
    pub default literal [](...items: T): Self from items {
        var list = Self.new()

        for item in items {
            list.push(move item)
        }

        return move list
    }
}

indexing.nct

struct Buffer {
    values: [i32; 3],
}

instance Buffer {
    operator (&self[index: usize]): &i32 {
        return &self.values[index]
    }

    operator (&+self[index: usize]): &+i32 {
        return &+self.values[index]
    }
}

func at<C>(container: &C, index: usize): i32 where (&C[usize]): &i32 {
    return container[index]
}

func main(): i32 {
    var buffer = Buffer { values: [3, 4, 5] }
    buffer[1] = 9
    if at(&buffer, 1) != 9 {
        return 1
    }
    return 0
}
