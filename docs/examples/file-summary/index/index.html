use std/io.print

func greet(user: User): void! {
    let name = user.display.name() otherwise { "guest" }

    let tone = match user.presence {
        Presence.online { "hello" }
        Presence.away { "welcome back" }
    }

    print("${tone}, ${name}\n")?
}
interface Writer {
    pub method &+self.write(bytes: &[u8]): void!
}

func save_log(output: &+Writer, events: &[Event]): void! {
    for event in events {
        output.write(event.message().bytes())?
        output.write("\n".bytes())?
    }
}
pub struct Session {
    id: SessionId
    user: User
    expires_at: Time
}

pub func Session.start(user: User, clock: &Clock): Session {
    return Session {
        id: SessionId.new(),
        user: move user,
        expires_at: clock.now().plus_minutes(30),
    }
}
func consume(values: Vec<String>): void! {
    for value in move values {
        print(&value as &str)?
    }
}
construct NonEmptyList<T> {
    pub default literal [](...items: T): Self from items {
        var list = Self.new()

        for item in items {
            list.push(move item)
        }

        return move list
    }
}

index.nct

use std/io.{open_path, print}

use std/num.usize_to_string

use std/path.Utf8Path

use std/process.arg

use ./summary

func main(): i32! {
    let requested: &str = arg(1)? otherwise {
        print("usage: file-summary PATH\n")?
        return 2
    }
    let path = Utf8Path.new(requested)?
    var file = open_path(&path)?
    let text = file.read_to_string()?
    let lines = count_lines(&text)
    let result = usize_to_string(lines)
    print((&result) as &str)?
    print("\n")?
    return 0
}
