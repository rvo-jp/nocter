func greet(user: User): void! {
    let name = user.display.name() otherwise { "guest" }

    let tone = match user.presence {
        Presence.online { "hello" }
        Presence.away { "welcome back" }
    }

    print("${tone}, ${name}\n")?
}
interface Writer {
    pub method &+self.write(bytes: &[u8]): void!
}

func save_log<W>(output: &+W, events: &[Event]): void! where W impl Writer {
    for event in events {
        output.write(event.message().bytes())?
        output.write("\n".bytes())?
    }
}

construct NonEmptyList<T> {
    pub literal [](...items: T): Self from items {
        var list = Self.new()

        for item in items {
            list.push(move item)
        }

        return move list
    }
}

use /internal/os.{OSError, OSErrorKind, Platform}
use /ptr.{addr, from_ref_mut}

#target: "arm64-darwin"
func errno(code: i32): Errno {
    return Errno { code: code }
}

#target: "arm64-darwin"
func error_from_errno(value: Errno): OSError {
    return OSError {
        platform: Platform.macos,
        code: value.code,
        kind: kind_from_errno(value),
    }
}

#target: "arm64-darwin"
func kind_from_errno(value: Errno): OSErrorKind {
    if value.code == 4 {
        return OSErrorKind.interrupted
    }
    if value.code == 35 {
        return OSErrorKind.would_block
    }
    if value.code == 2 {
        return OSErrorKind.not_found
    }
    if value.code == 20 {
        return OSErrorKind.not_directory
    }
    if value.code == 1 || value.code == 13 {
        return OSErrorKind.permission_denied
    }
    if value.code == 17 {
        return OSErrorKind.already_exists
    }
    if value.code == 9 || value.code == 14 || value.code == 21 || value.code == 22 {
        return OSErrorKind.invalid_input
    }
    if value.code == 32 {
        return OSErrorKind.broken_pipe
    }
    if value.code == 60 {
        return OSErrorKind.timed_out
    }
    if value.code == 45 {
        return OSErrorKind.unsupported
    }
    return OSErrorKind.unknown
}

func read_directory_entries(
    fd: usize,
    buffer: &+[u8],
    position: &+u64,
): SyscallResult {
    return syscall4(
        sys_getdirentries64,
        fd,
        addr(buffer.ptr()),
        buffer.len(),
        addr(from_ref_mut(position)),
    )
}
