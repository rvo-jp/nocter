use std/io

struct Point {
    x: i32
    y: i32
}

instance Point {
    impl Format
    method &self.format_into(output: &+String): void {
        output.push_str("(")
        self.x.format_into(output)
        output.push_str(", ")
        self.y.format_into(output)
        output.push_str(")")
        return
    }
}

func main(): i32! {
    let point = Point { x: 3, y: 4 }
    let message = "point = ${point}\n"
    io.print(&message)?
    return 0
}

use std/vec.Vec

struct Point {
    x: i32
    y: i32
}

instance Point {
    operator (&self == other: &Self): bool {
        return self.x == other.x && self.y == other.y
    }
}

func main(): i32! {
    let first = Point { x: 3, y: 4 }
    let same = Point { x: 3, y: 4 }
    let other = Point { x: 5, y: 6 }
    if first != same || first == other {
        return 1
    }
    let points = Vec [move first, move other]
    if points[0] != same {
        return 1
    }
    var values = Vec [1, 2, 3]
    values[1] = 4
    if values[0] + values[1] + values[2] != 8 {
        return 1
    }
    io.print("equality found the point\n")?
    return 0
}
struct Buffer {
    values: [i32; 3]
}

instance Buffer {
    operator (&self[index: usize]): &i32 {
        return &self.values[index]
    }

    operator (&+self[index: usize]): &+i32 {
        return &+self.values[index]
    }
}

func at<C>(container: &C, index: usize): i32 where (&C[usize]): &i32 {
    return container[index]
}

func main(): i32 {
    var buffer = Buffer { values: [3, 4, 5] }
    buffer[1] = 9
    if at(&buffer, 1) != 9 {
        return 1
    }
    return 0
}

#target: "arm64-darwin"
noalloc func errno(code: i32): Errno {
    return Errno { code: code }
}

#target: "arm64-darwin"
noalloc func error_from_errno(value: Errno): OSError {
    return OSError {
        platform: Platform.macos,
        code: value.code,
        kind: kind_from_errno(value),
    }
}

#target: "arm64-darwin"
noalloc func kind_from_errno(value: Errno): OSErrorKind {
    if value.code == 4 {
        return OSErrorKind.interrupted
    }
    if value.code == 35 {
        return OSErrorKind.would_block
    }
    if value.code == 2 {
        return OSErrorKind.not_found
    }
    if value.code == 20 {
        return OSErrorKind.not_directory
    }
    if value.code == 1 || value.code == 13 {
        return OSErrorKind.permission_denied
    }
    if value.code == 17 {
        return OSErrorKind.already_exists
    }
    if value.code == 9 || value.code == 14 || value.code == 21 || value.code == 22 {
        return OSErrorKind.invalid_input
    }
    if value.code == 32 {
        return OSErrorKind.broken_pipe
    }
    if value.code == 60 {
        return OSErrorKind.timed_out
    }
    if value.code == 45 {
        return OSErrorKind.unsupported
    }
    return OSErrorKind.unknown
}

#target: "arm64-darwin"
copy struct Timeval {
    seconds: u64
    microseconds: u64
}

#target: "arm64-darwin"
noalloc func read_directory_entries(
    fd: usize,
    buffer: &+[u8],
    position: &+u64,
): SyscallResult {
    return syscall4(
        SYS_GETDIRENTRIES64,
        fd,
        ptr.addr(buffer.ptr()),
        buffer.len(),
        ptr.addr(ptr.from_ref_mut(position)),
    )
}

#target: "arm64-darwin"
noalloc func wait_timeout_once(seconds: u64, microseconds: u64): SyscallResult {
    var timeout = Timeval { seconds: seconds, microseconds: microseconds }
    return syscall5(
        SYS_SELECT,
        0,
        0,
        0,
        0,
        ptr.addr(ptr.from_ref_mut(&+timeout)),
    )
}
