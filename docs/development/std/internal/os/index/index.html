use std/io.print

func greet(user: User): void! {
    let name = user.display.name() otherwise { "guest" }

    let tone = match user.presence {
        Presence.online { "hello" }
        Presence.away { "welcome back" }
    }

    print("${tone}, ${name}\n")?
}
interface Writer {
    pub method &+self.write(bytes: &[u8]): void!
}

func save_log<W>(output: &+W, events: &[Event]): void! where W: Writer {
    for event in events {
        output.write(event.message().bytes())?
        output.write("\n".bytes())?
    }
}

    }

func consume(values: Vec<String>): void! {
    for value in move values {
        print(&value as &str)?
    }
}
construct NonEmptyList<T> {
    pub default literal [](...items: T): Self from items {
        var list = Self.new()

        for item in items {
            list.push(move item)
        }

        return move list
    }
}

//! Common internal OS error model.
//!
//! Target-specific boundaries convert raw platform error values into OSError.
//! User-facing modules expose only the built-in `error` payload from fallible
//! APIs; OSError and OSErrorKind stay behind `pub(/)` visibility.

pub(/) enum Platform {
    macos
    linux
    windows
}

pub(/) enum OSErrorKind {
    interrupted
    would_block
    not_found
    permission_denied
    already_exists
    invalid_input
    broken_pipe
    timed_out
    unsupported
    unknown
}

pub(/) copy struct OSError {
    pub platform: Platform
    pub code: i32
    pub kind: OSErrorKind
}

#target: "arm64-darwin"
pub(/) copy struct SyscallResult {
    pub value: usize
    pub errno: i32
}

#target: "arm64-darwin"
pub(/) copy struct Errno {
    pub code: i32
}

#target: "arm64-darwin"
pub(/) func errno(code: i32): Errno {
    return Errno { code: code }
}

#target: "arm64-darwin"
pub(/) func error_from_errno(error: Errno): OSError {
    return OSError {
        platform: Platform.macos,
        code: error.code,
        kind: kind_from_errno(error),
    }
}

#target: "arm64-darwin"
pub(/) func kind_from_errno(error: Errno): OSErrorKind {
    // Darwin errno values used by the std syscall wrappers.
    if error.code == 4 {
        return OSErrorKind.interrupted
    }
    if error.code == 35 {
        return OSErrorKind.would_block
    }
    if error.code == 2 {
        return OSErrorKind.not_found
    }
    if error.code == 20 {
        return OSErrorKind.not_found
    }
    if error.code == 1 {
        return OSErrorKind.permission_denied
    }
    if error.code == 13 {
        return OSErrorKind.permission_denied
    }
    if error.code == 17 {
        return OSErrorKind.already_exists
    }
    if error.code == 9 {
        return OSErrorKind.invalid_input
    }
    if error.code == 14 {
        return OSErrorKind.invalid_input
    }
    if error.code == 21 {
        return OSErrorKind.invalid_input
    }
    if error.code == 22 {
        return OSErrorKind.invalid_input
    }
    if error.code == 32 {
        return OSErrorKind.broken_pipe
    }
    if error.code == 60 {
        return OSErrorKind.timed_out
    }
    if error.code == 45 {
        return OSErrorKind.unsupported
    }
    return OSErrorKind.unknown
}
