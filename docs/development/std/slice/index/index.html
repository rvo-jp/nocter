func greet(user: User): void! {
    let name = user.display.name() otherwise { "guest" }

    let tone = match user.presence {
        Presence.online { "hello" }
        Presence.away { "welcome back" }
    }

    print("${tone}, ${name}\n")?
}

}

func save_log(output: &+Writer, events: &[Event]): void! {
    for event in events {
        output.write(event.message().bytes())?
        output.write("\n".bytes())?
    }
}

//! Public observation methods for array slices.
//!
//! `[T]` is a compiler built-in unsized type. Its public behavior is ordinary
//! source built on the narrow representation query below.

use std/ptr.from_addr

pub(/) primitive slice_len_raw<T>(value: &[T]): usize
pub(/) primitive slice_ptr_addr_raw<T>(value: &[T]): usize

func values_equal<T>(left: &T, right: &T): bool where (&T == &T): bool {
    return left == right
}

func value_is_less<T>(left: &T, right: &T): bool where (&T < &T): bool {
    return left < right
}

instance [T] {
    pub method &self.len(): usize {
        return slice_len_raw(self)
    }

    pub method &self.is_empty(): bool {
        return slice_len_raw(self) == 0
    }

    pub method &self.ptr(): *T {
        return from_addr(slice_ptr_addr_raw(self))
    }

    /// Returns a readonly element borrow, or `none` when `index` is out of bounds.
    pub method &self.get(index: usize): &T? {
        if index >= self.len() {
            return none
        }
        return &self[index]
    }

    /// Returns a readwrite element borrow, or `none` when `index` is out of bounds.
    pub method &+self.get_mut(index: usize): &+T? {
        if index >= self.len() {
            return none
        }
        return &+self[index]
    }

    /// Returns the first element borrow, or `none` when the slice is empty.
    pub method &self.first(): &T? {
        return self.get(0)?
    }
}

instance [T] where (&T == &T): bool {
    /// Compares elements from left to right without consuming either slice.
    pub operator (&self == other: &Self): bool {
        if self.len() != other.len() {
            return false
        }
        var index: usize = 0
        while index < self.len() {
            if !values_equal(&self[index], &other[index]) {
                return false
            }
            index = index + 1
        }
        return true
    }

    /// Returns whether an equal element is present.
    pub method &self.contains(expected: &T): bool {
        let _index: usize = self.position(expected) otherwise { return false }
        return true
    }

    /// Returns the first index containing an equal element.
    pub method &self.position(expected: &T): usize? {
        var index: usize = 0
        while index < self.len() {
            if values_equal(&self[index], expected) {
                return index
            }
            index = index + 1
        }
        return none
    }
}

instance [T] where (&T < &T): bool {
    /// Orders elements lexicographically without consuming either slice.
    pub operator (&self < other: &Self): bool {
        let shared_len: usize = if self.len() < other.len() { self.len() } else { other.len() }
        var index: usize = 0
        while index < shared_len {
            if value_is_less(&self[index], &other[index]) { return true }
            if value_is_less(&other[index], &self[index]) { return false }
            index = index + 1
        }
        return self.len() < other.len()
    }
}
