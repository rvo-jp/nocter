struct Point {
    x: i32
    y: i32
}

instance Point {
    impl Format
    method &self.try_format_into(output: &+String): void! {
        output.try_push_str("(")?
        self.x.try_format_into(output)?
        output.try_push_str(", ")?
        self.y.try_format_into(output)?
        output.try_push_str(")")?
        return
    }
}

func main(): i32! {
    let point = Point { x: 3, y: 4 }
    let message = "point = ${point}\n"
    io.print(&message)?
    return 0
}
use std/io
use std/vec.Vec

struct Point {
    x: i32
    y: i32
}

instance Point {
    operator (&self == other: &Self): bool {
        return self.x == other.x && self.y == other.y
    }
}

func main(): i32! {
    let first = Point { x: 3, y: 4 }
    let same = Point { x: 3, y: 4 }
    let other = Point { x: 5, y: 6 }
    if first != same || first == other {
        return 1
    }
    let points = Vec [move first, move other]
    if points[0] != same {
        return 1
    }
    var values = Vec [1, 2, 3]
    values[1] = 4
    if values[0] + values[1] + values[2] != 8 {
        return 1
    }
    io.print("equality found the point\n")?
    return 0
}
struct Buffer {
    values: [i32; 3]
}

instance Buffer {
    operator (&self[index: usize]): &i32 {
        return &self.values[index]
    }

    operator (&+self[index: usize]): &+i32 {
        return &+self.values[index]
    }
}

func at<C>(container: &C, index: usize): i32 where (&C[usize]): &i32 {
    return container[index]
}

func main(): i32 {
    var buffer = Buffer { values: [3, 4, 5] }
    buffer[1] = 9
    if at(&buffer, 1) != 9 {
        return 1
    }
    return 0
}

command_darwin.nct

//! Darwin command orchestration and public failure selection.
//!
//! Raw target transitions expose typed facts. This layer is their only public-policy consumer: it
//! keeps prepared storage alive, closes parent descriptors, reaps every created child, and then
//! selects either an ExitStatus or one stable std.process failure.

use /internal/os.{OSErrorKind}
use /internal/os/darwin

#target: "arm64-darwin"
enum CommandFork {
    child(prepared: PreparedCommand, report: OwnedPipe)
    parent(prepared: PreparedCommand, pid: usize, report: OwnedPipe)
}

#target: "arm64-darwin"
noalloc func spawn_failed(): error {
    return error.new("std.process.spawn_failed", "child process creation failed")
}

#target: "arm64-darwin"
noalloc func wait_failed(): error {
    return error.new("std.process.wait_failed", "child process status could not be observed")
}

#target: "arm64-darwin"
noalloc func capture_failed(): error {
    return error.new("std.process.capture_failed", "child output could not be captured")
}

#target: "arm64-darwin"
noalloc func exec_failed(errno: i32): error {
    let os_error = darwin.error_from_errno(darwin.errno(errno))
    if os_error.kind is OSErrorKind.not_found {
        return error.new("std.process.not_found", "executable was not found")
    }
    if os_error.kind is OSErrorKind.not_directory {
        return error.new("std.process.not_found", "executable was not found")
    }
    if os_error.kind is OSErrorKind.permission_denied {
        return error.new("std.process.permission_denied", "executable permission was denied")
    }
    if os_error.kind is OSErrorKind.invalid_input {
        return error.new("std.process.invalid_input", "executable request was invalid")
    }
    return spawn_failed()
}

#target: "arm64-darwin"
noalloc func status_from_wait(outcome: WaitOutcome): ExitStatus! {
    match move outcome {
        WaitOutcome.exited(code) { return exited_status(code) }
        WaitOutcome.signaled(signal) { return signaled_status(signal) }
        WaitOutcome.failed(_) { return wait_failed() }
        WaitOutcome.invalid(_) { return wait_failed() }
    }
}

#target: "arm64-darwin"
func observe_terminal(prepared: PreparedCommand, pid: usize): ExitStatus! {
    let terminal = status_from_wait(wait_for_child(pid))?
    drop prepared
    return terminal
}

#target: "arm64-darwin"
func observe_parent(
    prepared: PreparedCommand,
    pid: usize,
    report: OwnedPipe,
): ExitStatus! {
    let launch = read_launch_report(move report)

    // Waiting precedes public failure selection so every successfully created child is reaped,
    // including children whose exec request was rejected or whose report channel was malformed.
    let terminal = observe_terminal(move prepared, pid)?
    match move launch {
        LaunchReportAttempt.executed { return terminal }
        LaunchReportAttempt.setup_rejected(_) { return spawn_failed() }
        LaunchReportAttempt.exec_rejected(errno) { return exec_failed(errno) }
        LaunchReportAttempt.channel_failed(_) { return spawn_failed() }
        LaunchReportAttempt.malformed { return spawn_failed() }
    }
}

#target: "arm64-darwin"
func observe_output_parent(
    prepared: PreparedCommand,
    pid: usize,
    report: OwnedPipe,
    captures: CaptureSession,
): Output! {
    let launch = read_launch_report(move report)
    let captured = drain_captured_streams(move captures)

    // Waiting remains the last fallible lifecycle transition and precedes public failure
    // selection. A wait failure therefore cannot be hidden by an earlier launch or capture error.
    let terminal = observe_terminal(move prepared, pid)?
    match move launch {
        LaunchReportAttempt.setup_rejected(_) { return capture_failed() }
        LaunchReportAttempt.exec_rejected(errno) { return exec_failed(errno) }
        LaunchReportAttempt.channel_failed(_) { return spawn_failed() }
        LaunchReportAttempt.malformed { return spawn_failed() }
        LaunchReportAttempt.executed {
            match move captured {
                CaptureSessionAttempt.complete(stdout, stderr) {
                    return Output {
                        status: terminal,
                        stdout: move stdout,
                        stderr: move stderr,
                    }
                }
                CaptureSessionAttempt.failed(_) { return capture_failed() }
                CaptureSessionAttempt.malformed { return capture_failed() }
            }
        }
    }
}

#target: "arm64-darwin"
func fork_prepared_command(command: Command): CommandFork! {
    let prepared = prepare_command(move command)
    match create_cloexec_pipe() {
        PipeAttempt.failed(_) { return spawn_failed() }
        PipeAttempt.ready(report) {
            match fork_process() {
                ForkAttempt.failed(_) {
                    drop report
                    return spawn_failed()
                }
                ForkAttempt.child {
                    return CommandFork.child(move prepared, move report)
                }
                ForkAttempt.parent(pid) {
                    return CommandFork.parent(move prepared, pid, move report)
                }
            }
        }
    }
}

#target: "arm64-darwin"
func run_command(command: Command): ExitStatus! {
    match fork_prepared_command(move command)? {
        CommandFork.child(prepared, report) {
            exec_child(
                prepared_path_address(&prepared),
                prepared_argv_address(&prepared),
                env_vector_raw(),
                move report,
            )
        }
        CommandFork.parent(prepared, pid, report) {
            return observe_parent(move prepared, pid, move report)?
        }
    }
}

#target: "arm64-darwin"
func output_command(command: Command): Output! {
    match create_capture_session() {
        CapturePreparationAttempt.failed(_) { return capture_failed() }
        CapturePreparationAttempt.ready(captures) {
            match fork_prepared_command(move command)? {
                CommandFork.child(prepared, report) {
                    exec_captured_child(
                        prepared_path_address(&prepared),
                        prepared_argv_address(&prepared),
                        env_vector_raw(),
                        move report,
                        move captures,
                    )
                }
                CommandFork.parent(prepared, pid, report) {
                    return observe_output_parent(
                        move prepared,
                        pid,
                        move report,
                        move captures,
                    )?
                }
            }
        }
    }
}

instance Command {
    method self.status(): ExitStatus! {
        return run_command(move self)?
    }

    method self.output(): Output! {
        return output_command(move self)?
    }
}
