#target: "arm64-darwin"
func open_current_directory(): usize! {
    let path = target_path_argument(".")?
    let result = syscall3(sys_open, target_path_address(&path), 0, 0)
    if result.errno != 0 {
        return cwd_failed()
    }
    return result.value
}

#target: "arm64-darwin"
func close_fd_word(fd: usize): void {
    let result = syscall1(sys_close, fd)
    return
}

#target: "arm64-darwin"
func cwd_path_len(buffer: &+RawBuffer): usize! {
    let buffer_len = buffer.bytes().len()
    let raw_bytes = buffer.bytes_mut()
    var index: usize = 0
    while index < buffer_len {
        if raw_bytes[index] == 0 {
            return index
        }
        index += 1
    }
    return cwd_failed()
}

#target: "arm64-darwin"
func fill_cwd_path(fd: usize, buffer: &+RawBuffer): usize! {
    let result = syscall3(
        sys_fcntl,
        fd,
        f_getpath,
        addr(buffer.bytes().ptr()),
    )
    if result.errno != 0 {
        return cwd_failed()
    }
    return cwd_path_len(buffer)?
}

func args(): Vec<&str>! {
    let count: usize = arg_count_raw()
    var values: Vec<&str> = Vec.with_capacity(count)
    var index: usize = 0
    while index < count {
        let value = arg_raw(index)
        if !is_valid_utf8(value.bytes()) {
            return invalid_encoding()
        }
        values.push(value)
        index += 1
    }
    return move values
}

func arg_count(): usize {
    return arg_count_raw()
}

func arg(index: usize): &str?! {
    let count: usize = arg_count_raw()
    if index >= count { return none }
    let value = arg_raw(index)
    if !is_valid_utf8(value.bytes()) { return invalid_encoding() }
    return value
}

func environment_count(): usize {
    return env_count_raw()
}

func environment(index: usize): EnvironmentEntry?! {
    let count: usize = env_count_raw()
    if index >= count { return none }
    let name = env_name_raw(index)
    let value = env_value_raw(index)
    if !is_valid_utf8(name.bytes()) || !is_valid_utf8(value.bytes()) { return invalid_encoding() }
    return EnvironmentEntry { name: name, value: value }
}

func env(name: &str): &str?! {
    if !valid_environment_name(name) {
        return invalid_encoding()
    }
    let count = env_count_raw()
    var index: usize = 0
    while index < count {
        let entry_name = env_name_raw(index)
        let entry_value = env_value_raw(index)
        if !is_valid_utf8(entry_name.bytes()) || !is_valid_utf8(entry_value.bytes()) {
            return invalid_encoding()
        }
        if entry_name == name {
            return entry_value
        }
        index += 1
    }
    return none
}

func cwd(): String! {
    let fd = open_current_directory()?
    var allocator = page_allocator()
    var buffer = allocator.alloc(max_path_len, 1)
    let len = fill_cwd_path(fd, &+buffer) catch failure {
        drop buffer
        close_fd_word(fd)
        return move failure
    }
    let view = str_from_raw_parts(buffer.bytes().ptr(), len)
    let value = String.copy(view)
    drop buffer
    close_fd_word(fd)
    return move value
}

func try_cwd(allocator: &+TryAllocator): String! {
    let fd = open_current_directory() catch failure {
        return move failure
    }
    var buffer = allocator.try_alloc(max_path_len, 1) catch failure {
        close_fd_word(fd)
        return move failure
    }
    let len = fill_cwd_path(fd, &+buffer) catch failure {
        drop buffer
        close_fd_word(fd)
        return move failure
    }
    let view = str_from_raw_parts(buffer.bytes().ptr(), len)
    let value = String.try_copy(allocator, view) catch failure {
        drop buffer
        close_fd_word(fd)
        return move failure
    }
    drop buffer
    close_fd_word(fd)
    return move value
}

func valid_environment_name(name: &str): bool {
    if name.len() == 0 || !is_valid_utf8(name.bytes()) {
        return false
    }
    let name_bytes = name.bytes()
    var index: usize = 0
    while index < name_bytes.len() {
        if name_bytes[index] == 0 || name_bytes[index] == 61 {
            return false
        }
        index += 1
    }
    return true
}

func exit(code: i32): never {
    exit_raw(code)
}

func abort(): never {
    trap()
}

func cwd_failed(): error {
    return error.new("std.process.cwd_failed", "current working directory is not available")
}

func invalid_encoding(): error {
    return error.new(
        "std.process.invalid_encoding",
        "process text or requested environment name is invalid",
    )
}
