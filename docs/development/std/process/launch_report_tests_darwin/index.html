use std/io

struct Point {
    x: i32
    y: i32
}

instance Point {
    impl Format
    method &self.try_format_into(output: &+String): void! {
        output.try_push_str("(")?
        self.x.try_format_into(output)?
        output.try_push_str(", ")?
        self.y.try_format_into(output)?
        output.try_push_str(")")?
        return
    }
}

func main(): i32! {
    let point = Point { x: 3, y: 4 }
    let message = "point = ${point}\n"
    io.print(&message)?
    return 0
}
use std/io
use std/vec.Vec

struct Point {
    x: i32
    y: i32
}

instance Point {
    operator (&self == other: &Self): bool {
        return self.x == other.x && self.y == other.y
    }
}

func main(): i32! {
    let first = Point { x: 3, y: 4 }
    let same = Point { x: 3, y: 4 }
    let other = Point { x: 5, y: 6 }
    if first != same || first == other {
        return 1
    }
    let points = Vec [move first, move other]
    if points[0] != same {
        return 1
    }
    var values = Vec [1, 2, 3]
    values[1] = 4
    if values[0] + values[1] + values[2] != 8 {
        return 1
    }
    io.print("equality found the point\n")?
    return 0
}
struct Buffer {
    values: [i32; 3]
}

instance Buffer {
    operator (&self[index: usize]): &i32 {
        return &self.values[index]
    }

    operator (&+self[index: usize]): &+i32 {
        return &+self.values[index]
    }
}

func at<C>(container: &C, index: usize): i32 where (&C[usize]): &i32 {
    return container[index]
}

func main(): i32 {
    var buffer = Buffer { values: [3, 4, 5] }
    buffer[1] = 9
    if at(&buffer, 1) != 9 {
        return 1
    }
    return 0
}

launch_report_tests_darwin.nct

//! Native qualification for the staged close-on-exec launch report and public failure policy.

see ./command_darwin.nct
see ./launch_report_darwin.nct
see ./pipe_darwin.nct

use /internal/os/darwin
use /ptr

#target: "arm64-darwin"
func authored_launch_report(
    stage: i32,
    errno: i32,
    byte_count: usize,
): LaunchReportAttempt! {
    match create_cloexec_pipe() {
        PipeAttempt.failed(_) {
            return error.new("std.process.test", "launch report pipe creation failed")
        }
        PipeAttempt.ready(pipe) {
            var channel = move pipe
            let payload = LaunchFailurePayload { stage: stage, errno: errno }
            let result = darwin.syscall3(
                darwin.SYS_WRITE,
                pipe_write_fd(&channel),
                ptr.addr(ptr.from_ref(&payload)),
                byte_count,
            )
            close_pipe_write(&+channel)
            if result.errno != 0 || result.value != byte_count {
                return error.new("std.process.test", "launch report write failed")
            }
            return read_launch_report(move channel)
        }
    }
}

#target: "arm64-darwin"
func closed_launch_report(): LaunchReportAttempt! {
    match create_cloexec_pipe() {
        PipeAttempt.failed(_) {
            return error.new("std.process.test", "closed report pipe creation failed")
        }
        PipeAttempt.ready(pipe) {
            var channel = move pipe
            close_pipe_write(&+channel)
            return read_launch_report(move channel)
        }
    }
}

#target: "arm64-darwin"
func failed_launch_report(): LaunchReportAttempt! {
    match create_cloexec_pipe() {
        PipeAttempt.failed(_) {
            return error.new("std.process.test", "failed report pipe creation failed")
        }
        PipeAttempt.ready(pipe) {
            var channel = move pipe
            close_pipe_read(&+channel)
            return read_launch_report(move channel)
        }
    }
}

#target: "arm64-darwin"
noalloc func expect_launch_failure(
    attempt: LaunchReportAttempt,
    expected_stage: i32,
    expected_errno: i32,
): bool {
    match move attempt {
        LaunchReportAttempt.input_setup_rejected(errno) {
            return expected_stage == LAUNCH_STAGE_INPUT_SETUP && errno == expected_errno
        }
        LaunchReportAttempt.output_setup_rejected(errno) {
            return expected_stage == LAUNCH_STAGE_OUTPUT_SETUP && errno == expected_errno
        }
        LaunchReportAttempt.directory_rejected(errno) {
            return expected_stage == LAUNCH_STAGE_DIRECTORY && errno == expected_errno
        }
        LaunchReportAttempt.exec_rejected(errno) {
            return expected_stage == LAUNCH_STAGE_EXEC && errno == expected_errno
        }
        _ { return false }
    }
}

#target: "arm64-darwin"
noalloc func expect_malformed_launch_report(attempt: LaunchReportAttempt): bool {
    match move attempt {
        LaunchReportAttempt.malformed { return true }
        _ { return false }
    }
}

#target: "arm64-darwin"
func launch_failure_has_public_code(attempt: LaunchReportAttempt, code: &str): bool {
    let _ = require_successful_launch(move attempt) catch failure {
        return failure.has_code(code)
    }
    return false
}

test launch_report_preserves_every_stage_protocol_and_public_policy {
    match closed_launch_report()? {
        LaunchReportAttempt.executed {}
        _ { return error.new("std.process.test", "closed report did not prove exec") }
    }

    if !expect_launch_failure(
        authored_launch_report(LAUNCH_STAGE_INPUT_SETUP, 9, LAUNCH_REPORT_BYTES)?,
        LAUNCH_STAGE_INPUT_SETUP,
        9,
    ) || !expect_launch_failure(
        authored_launch_report(LAUNCH_STAGE_OUTPUT_SETUP, 10, LAUNCH_REPORT_BYTES)?,
        LAUNCH_STAGE_OUTPUT_SETUP,
        10,
    ) || !expect_launch_failure(
        authored_launch_report(LAUNCH_STAGE_DIRECTORY, 11, LAUNCH_REPORT_BYTES)?,
        LAUNCH_STAGE_DIRECTORY,
        11,
    ) || !expect_launch_failure(
        authored_launch_report(LAUNCH_STAGE_EXEC, 13, LAUNCH_REPORT_BYTES)?,
        LAUNCH_STAGE_EXEC,
        13,
    ) {
        return error.new("std.process.test", "launch report stage or errno changed")
    }

    if !expect_malformed_launch_report(
        authored_launch_report(99, 13, LAUNCH_REPORT_BYTES)?,
    ) || !expect_malformed_launch_report(
        authored_launch_report(LAUNCH_STAGE_EXEC, 0, LAUNCH_REPORT_BYTES)?,
    ) || !expect_malformed_launch_report(
        authored_launch_report(LAUNCH_STAGE_EXEC, 13, 2)?,
    ) {
        return error.new("std.process.test", "invalid report became a launch fact")
    }

    match failed_launch_report()? {
        LaunchReportAttempt.channel_failed(errno) {
            if errno == 0 { return error.new("std.process.test", "channel failure lost errno") }
        }
        _ { return error.new("std.process.test", "failed report channel changed classification") }
    }

    if !launch_failure_has_public_code(
        LaunchReportAttempt.input_setup_rejected(9),
        "std.process.input_failed",
    ) || !launch_failure_has_public_code(
        LaunchReportAttempt.output_setup_rejected(9),
        "std.process.capture_failed",
    ) || !launch_failure_has_public_code(
        LaunchReportAttempt.directory_rejected(9),
        "std.process.current_directory_failed",
    ) || !launch_failure_has_public_code(
        LaunchReportAttempt.channel_failed(9),
        "std.process.spawn_failed",
    ) {
        return error.new("std.process.test", "launch stage changed public failure policy")
    }
    return
}
