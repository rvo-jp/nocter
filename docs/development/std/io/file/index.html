//! File ownership and Darwin descriptor operations.

see ./index.nct

use /internal/os/darwin.{SyscallResult, create_mode, errno_interrupted}
use /internal/os/darwin.{open_create_append, open_create_truncate, open_read_only}
use /internal/os/darwin.{sys_close, sys_open, sys_read, sys_write}
use /internal/io.{closed_stream_error, io_error_from_os}
use /internal/os/darwin.{errno, error_from_errno, syscall1, syscall3}
use /internal/path.{target_path_address, target_path_argument}
use /ptr.addr
use /string.String
use /vec.Vec

#target: "arm64-darwin"
const stdout_fd: usize = 1

#target: "arm64-darwin"
const stderr_fd: usize = 2

#target: "arm64-darwin"
func open_read_fd(path: &str): usize! {
    return open_fd(path, open_read_only)?
}

#target: "arm64-darwin"
func create_fd(path: &str): usize! {
    return open_fd(path, open_create_truncate)?
}

#target: "arm64-darwin"
func append_fd(path: &str): usize! {
    return open_fd(path, open_create_append)?
}

#target: "arm64-darwin"
func open_fd(path: &str, flags: usize): usize! {
    let argument = target_path_argument(path)?

    var result: SyscallResult = syscall3(
        sys_open,
        target_path_address(&argument),
        flags,
        create_mode,
    )
    while result.errno == errno_interrupted {
        result = syscall3(
            sys_open,
            target_path_address(&argument),
            flags,
            create_mode,
        )
    }
    if result.errno != 0 {
        return io_error_from_os(error_from_errno(errno(result.errno)))
    }
    return result.value
}

#target: "arm64-darwin"
func read_fd(fd: usize, buffer: &+[u8]): usize! {
    var result: SyscallResult = syscall3(
        sys_read,
        fd,
        addr(buffer.ptr()),
        buffer.len(),
    )
    while result.errno == errno_interrupted {
        result = syscall3(sys_read, fd, addr(buffer.ptr()), buffer.len())
    }
    if result.errno != 0 {
        return io_error_from_os(error_from_errno(errno(result.errno)))
    }
    if result.value > buffer.len() {
        return invalid_read_count()
    }
    return result.value
}

#target: "arm64-darwin"
func write_fd(fd: usize, bytes: &[u8]): void! {
    var offset: usize = 0
    while offset < bytes.len() {
        let remaining: usize = bytes.len() - offset
        var result: SyscallResult = syscall3(
            sys_write,
            fd,
            addr(bytes.ptr()) + offset,
            remaining,
        )
        while result.errno == errno_interrupted {
            result = syscall3(
                sys_write,
                fd,
                addr(bytes.ptr()) + offset,
                remaining,
            )
        }
        if result.errno != 0 {
            return io_error_from_os(error_from_errno(errno(result.errno)))
        }
        if result.value == 0 {
            return write_zero()
        }
        if result.value > remaining {
            return invalid_write_count()
        }
        offset += result.value
    }
    return
}

#target: "arm64-darwin"
func write_text_fd(fd: usize, text: &str): void! {
    write_fd(fd, text.bytes())?
    return
}

#target: "arm64-darwin"
func close_fd(fd: usize): void {
    // A failed close is deliberately not retried: Darwin may have consumed the
    // descriptor even when reporting an interruption, and retrying could close
    // an unrelated descriptor that another operation has reused.
    let _ = syscall1(sys_close, fd)
    return
}

enum FileState {
    owned_open
    borrowed_open
    closed
}

struct File {
    fd: usize
    state: FileState
}

func stdout(): File {
    return File { fd: stdout_fd, state: FileState.borrowed_open }
}

func stderr(): File {
    return File { fd: stderr_fd, state: FileState.borrowed_open }
}

construct File {
    func open(path: &str): Self! {
        let fd = open_read_fd(path)?
        return File { fd: fd, state: FileState.owned_open }
    }

    func create(path: &str): Self! {
        let fd = create_fd(path)?
        return File { fd: fd, state: FileState.owned_open }
    }

    func append(path: &str): Self! {
        let fd = append_fd(path)?
        return File { fd: fd, state: FileState.owned_open }
    }
}

func require_open(file: &File): void! {
    if file.state is FileState.closed {
        return closed_stream_error()
    }
    return
}

func read(file: &+File, buffer: &+[u8]): usize! {
    require_open(file)?
    return read_fd(file.fd, buffer)?
}

func write(file: &+File, bytes: &[u8]): void! {
    require_open(file)?
    write_fd(file.fd, bytes)?
    return
}

func close(file: &+File): void {
    if file.state is FileState.owned_open {
        close_fd(file.fd)
    }
    file.state = FileState.closed
    return
}

instance File {
    method &+self.close(): void {
        close(self)
        return
    }

    method &+self.read(buffer: &+[u8]): usize! {
        return read(self, buffer)?
    }

    method &+self.write(bytes: &[u8]): void! {
        write(self, bytes)?
        return
    }

    method &+self.flush(): void! {
        require_open(self)?
        return
    }
}

drop File(&+self) {
    close(self)
    return
}

func print(text: &str): void! {
    write_text_fd(1, text)?
    return
}

func write_zero(): error {
    return error.new("std.io.write_zero", "writer made no progress")
}

func invalid_write_count(): error {
    return error.new(
        "std.io.invalid_write_count",
        "writer reported more bytes than the supplied input contains",
    )
}
