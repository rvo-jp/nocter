use std/vec.Vec

struct Point {
    x: i32
    y: i32
}

instance Point {
    operator (&self == other: &Self): bool {
        return self.x == other.x && self.y == other.y
    }
}

func main(): i32! {
    let first = Point { x: 3, y: 4 }
    let same = Point { x: 3, y: 4 }
    let other = Point { x: 5, y: 6 }
    if first != same || first == other {
        return 1
    }
    let points = Vec [move first, move other]
    if points[0] != same {
        return 1
    }
    var values = Vec [1, 2, 3]
    values[1] = 4
    if values[0] + values[1] + values[2] != 8 {
        return 1
    }
    io.print("equality found the point\n")?
    return 0
}

func at<C>(container: &C, index: usize): i32 where (&C[usize]): &i32 {
    return container[index]
}

func main(): i32 {
    var buffer = Buffer { values: [3, 4, 5] }
    buffer[1] = 9
    if at(&buffer, 1) != 9 {
        return 1
    }
    return 0

//! Buffered reader and writer state machines.

see ./index.nct

use /io.{File, Reader, Writer}
use /internal/io
use /string.String
use /vec.Vec

struct BufReader {
    source: File
    buffer: Vec<u8>
    next: usize
    line_buffer: Vec<u8>
    is_finished: bool
}

construct BufReader {
    func new(source: File): Self {
        return BufReader.with_capacity(move source, 8192)
    }

    func with_capacity(source: File, requested_capacity: usize): Self {
        var capacity = requested_capacity
        if capacity == 0 {
            capacity = 1
        }
        let storage: Vec<u8> = Vec.with_capacity(capacity)
        return BufReader {
            source: move source,
            buffer: move storage,
            next: 0,
            line_buffer: Vec.empty(),
            is_finished: false,
        }
    }
}

func read(reader: &+BufReader, destination: &+[u8]): usize! {
    if reader.is_finished {
        return 0
    }
    var written: usize = 0
    while written < destination.len() {
        if reader.next == reader.buffer.len() {
            let received = refill(reader)?
            if received == 0 {
                break
            }
        }
        let available: &[u8] = &reader.buffer
        destination[written] = available[reader.next]
        written += 1
        reader.next += 1
    }
    return written
}

func refill(reader: &+BufReader): usize! {
    if reader.is_finished {
        return 0
    }
    reader.buffer.clear()
    var initialized: usize = 0
    while initialized < reader.buffer.capacity() {
        reader.buffer.push(0)
        initialized += 1
    }
    let received = reader.source.read(&+reader.buffer) catch failure {
        terminate_reader(reader)
        return move failure
    }
    reader.buffer.truncate(received)
    reader.next = 0
    if received == 0 {
        end_reader(reader)
    }
    return received
}

func read_line(reader: &+BufReader): String?! {
    var result = String.empty()
    let present = read_line_into(reader, &+result)?
    if !present {
        return none
    }
    return move result
}

func read_line_into(reader: &+BufReader, destination: &+String): bool! {
    destination.clear()
    if reader.is_finished {
        return false
    }
    reader.line_buffer.clear()

    while true {
        if reader.next == reader.buffer.len() {
            let received = refill(reader)?
            if received == 0 {
                if reader.line_buffer.len() == 0 {
                    return false
                }
                return commit_line(reader, destination)?
            }
        }

        let available: &[u8] = &reader.buffer
        let byte = available[reader.next]
        reader.next += 1
        if byte == 10 {
            let line: &[u8] = &reader.line_buffer
            let ends_with_cr = line.len() != 0 && line[line.len() - 1] == 13
            if ends_with_cr {
                let _ = reader.line_buffer.pop()
            }
            return commit_line(reader, destination)?
        }
        reader.line_buffer.try_push(byte) catch failure {
            terminate_reader(reader)
            return move failure
        }
    }
    return false
}

func commit_line(reader: &+BufReader, destination: &+String): bool! {
    write_validated_line(&reader.line_buffer, destination) catch failure {
        terminate_reader(reader)
        return move failure
    }
    reader.line_buffer.clear()
    return true
}

func write_validated_line(line: &Vec<u8>, destination: &+String): void! {
    let bytes: &[u8] = line
    destination.try_push_utf8(bytes)?
    return
}

func end_reader(reader: &+BufReader): void {
    if !reader.is_finished {
        reader.source.close()
        reader.is_finished = true
    }
    reader.buffer.clear()
    reader.next = 0
    return
}

func terminate_reader(reader: &+BufReader): void {
    end_reader(reader)
    reader.line_buffer.clear()
    return
}

instance BufReader {
    method &+self.read_line(): String?! {
        return read_line(self)
    }

    method &+self.read_line_into(destination: &+String): bool! {
        return read_line_into(self, destination)?
    }

    method &+self.close(): void {
        terminate_reader(self)
        return
    }

    method &+self.read(destination: &+[u8]): usize! {
        return read(self, destination)?
    }
}

struct BufWriter {
    destination: File
    buffer: Vec<u8>
    limit: usize
    is_open: bool
}

construct BufWriter {
    func new(destination: File): Self {
        return BufWriter.with_capacity(move destination, 8192)
    }

    func with_capacity(destination: File, requested_capacity: usize): Self {
        var capacity = requested_capacity
        if capacity == 0 {
            capacity = 1
        }
        let storage: Vec<u8> = Vec.with_capacity(capacity)
        return BufWriter {
            destination: move destination,
            buffer: move storage,
            limit: capacity,
            is_open: true,
        }
    }
}

func write(writer: &+BufWriter, bytes: &[u8]): void! {
    if !writer.is_open {
        return io.closed_stream_error()
    }
    var offset: usize = 0
    while offset < bytes.len() {
        if writer.buffer.len() == writer.limit {
            flush(writer)?
        }
        writer.buffer.push(bytes[offset])
        offset += 1
    }
    return
}

func flush(writer: &+BufWriter): void! {
    if !writer.is_open {
        return io.closed_stream_error()
    }
    if writer.buffer.len() == 0 {
        return
    }
    writer.destination.write(&writer.buffer) catch failure {
        terminate_writer(writer)
        return move failure
    }
    writer.buffer.clear()
    writer.destination.flush() catch failure {
        terminate_writer(writer)
        return move failure
    }
    return
}

func close(writer: &+BufWriter): void! {
    if !writer.is_open {
        return
    }
    flush(writer)?
    terminate_writer(writer)
    return
}

func terminate_writer(writer: &+BufWriter): void {
    writer.destination.close()
    writer.buffer.clear()
    writer.is_open = false
    return
}

instance BufWriter {
    method &+self.close(): void! {
        close(self)?
        return
    }

    method &+self.write(bytes: &[u8]): void! {
        write(self, bytes)?
        return
    }

    method &+self.flush(): void! {
        flush(self)?
        return
    }
}
