use std/vec.Vec

struct Point {
    x: i32
    y: i32
}

instance Point {
    operator (&self == other: &Self): bool {
        return self.x == other.x && self.y == other.y
    }
}

func main(): i32! {
    let first = Point { x: 3, y: 4 }
    let same = Point { x: 3, y: 4 }
    let other = Point { x: 5, y: 6 }
    if first != same || first == other {
        return 1
    }
    let points = Vec [move first, move other]
    if points[0] != same {
        return 1
    }
    var values = Vec [1, 2, 3]
    values[1] = 4
    if values[0] + values[1] + values[2] != 8 {
        return 1
    }
    io.print("equality found the point\n")?
    return 0
}
struct Buffer {
    values: [i32; 3]
}

instance Buffer {
    operator (&self[index: usize]): &i32 {
        return &self.values[index]
    }

    operator (&+self[index: usize]): &+i32 {
        return &+self.values[index]
    }
}

func at<C>(container: &C, index: usize): i32 where (&C[usize]): &i32 {
    return container[index]
}

func main(): i32 {
    var buffer = Buffer { values: [3, 4, 5] }
    buffer[1] = 9
    if at(&buffer, 1) != 9 {
        return 1
    }
    return 0
}

#target: "arm64-darwin"
func alloc_pages(size: usize): usize! {
    let result = darwin.syscall6(
        darwin.SYS_MMAP,
        0,
        size,
        darwin.MMAP_READ_WRITE,
        darwin.MMAP_PRIVATE_ANONYMOUS,
        darwin.MMAP_NO_FILE,
        0,
    )
    if result.errno != 0 {
        return allocation_failure_error()
    }
    return result.value
}

#target: "arm64-darwin"
noalloc func free_pages(address: usize, size: usize): void {
    let result = darwin.syscall2(darwin.SYS_MUNMAP, address, size)
    return
}

func region_alloc_pages(
    state: usize,
    size: usize,
    align: usize,
): usize! {
    // Every mapping begins with a release-list header. The payload is aligned
    // within the mapping, so region release can walk mappings without storing
    // metadata beside or inside user-visible bytes.
    let header_size: usize = 16
    let header_and_payload: usize = mem.checked_add(header_size, size) otherwise {
        return invalid_argument()
    }
    let mapping_size: usize = mem.checked_add(header_and_payload, align) otherwise {
        return invalid_argument()
    }
    let mapping = alloc_pages(mapping_size)?
    let payload_start = mapping + header_size
    let remainder = payload_start % align
    var padding: usize = 0
    if remainder != 0 {
        padding = align - remainder
    }

    let state_pointer: *usize = ptr.from_addr(state)
    let state_words: &[usize] = ptr.slice_from_raw_parts_value(state_pointer, 1)
    let previous: usize = state_words[0]
    let header: *usize = ptr.from_addr(mapping)
    ptr.store_value_to_ptr(header, 0, previous)
    ptr.store_value_to_ptr(header, 8, mapping_size)
    ptr.store_value_to_ptr(state_pointer, 0, mapping)
    return payload_start + padding
}

func try_alloc(
    allocator: &+TryAllocator,
    size: usize,
    align: usize,
): RawBuffer! {
    let requested = Layout.new(size, align)?
    return try_alloc_layout(allocator, requested)?
}

func try_alloc_layout(
    allocator: &+TryAllocator,
    requested: Layout,
): RawBuffer! {
    if requested.size == 0 {
        return RawBuffer {
            ptr: ptr.from_addr(1),
            len: 0,
            align: requested.align,
            allocator_state: allocator.state,
            allocator_kind: allocator.kind,
        }
    }

    var address: usize = 0
    if allocator_kind_is_page(allocator.kind) {
        address = alloc_pages(requested.size)?
    } else if allocator_kind_is_region(allocator.kind) {
        address = region_alloc_pages(allocator.state, requested.size, requested.align)?
    } else {
        return invalid_argument()
    }
    return RawBuffer {
        ptr: ptr.from_addr(address),
        len: requested.size,
        align: requested.align,
        allocator_state: allocator.state,
        allocator_kind: allocator.kind,
    }
}

func alloc(
    allocator: &+Allocator,
    size: usize,
    align: usize,
): RawBuffer {
    var fallible = TryAllocator { state: allocator.state, kind: allocator.kind }
    return try_alloc(&+fallible, size, align) catch allocation_error {
        return mem.allocation_abort()
    }
}

func alloc_layout(
    allocator: &+Allocator,
    requested: Layout,
): RawBuffer {
    var fallible = TryAllocator { state: allocator.state, kind: allocator.kind }
    return try_alloc_layout(&+fallible, requested) catch allocation_error {
        return mem.allocation_abort()
    }
}

func try_grow(
    allocator: &+TryAllocator,
    buffer: &+RawBuffer,
    new_size: usize,
): void! {
    if new_size < buffer.len {
        return invalid_argument()
    }
    if new_size == buffer.len {
        return
    }
    if !allocator_kind_is_unbound(buffer.allocator_kind) {
        if allocator.state != buffer.allocator_state {
            return invalid_argument()
        }
        if allocator.kind != buffer.allocator_kind {
            return invalid_argument()
        }
    }

    var replacement = try_alloc(allocator, new_size, buffer.align)?
    if buffer.len != 0 {
        ptr.copy_ptr_to_ptr(replacement.ptr, buffer.ptr, buffer.len)
    }

    var previous = RawBuffer {
        ptr: buffer.ptr,
        len: buffer.len,
        align: buffer.align,
        allocator_state: buffer.allocator_state,
        allocator_kind: buffer.allocator_kind,
    }

    buffer.ptr = replacement.ptr
    buffer.len = replacement.len
    buffer.align = replacement.align
    buffer.allocator_state = replacement.allocator_state
    buffer.allocator_kind = replacement.allocator_kind

    replacement.ptr = ptr.from_addr(1)
    replacement.len = 0
    return
}

func try_grow_owned(buffer: &+RawBuffer, new_size: usize): void! {
    var allocator = TryAllocator {
        state: buffer.allocator_state,
        kind: buffer.allocator_kind,
    }
    if allocator_kind_is_unbound(buffer.allocator_kind) {
        allocator.state = current_allocator_state()
        allocator.kind = current_allocator_kind()
    }
    try_grow(&+allocator, buffer, new_size)?
    return
}

func grow(allocator: &+Allocator, buffer: &+RawBuffer, new_size: usize): void {
    var fallible = TryAllocator { state: allocator.state, kind: allocator.kind }
    try_grow(&+fallible, buffer, new_size) catch allocation_error {
        return mem.allocation_abort()
    }
    return
}

instance Allocator {
    method &+self.alloc(size: usize, align: usize): RawBuffer {
        return alloc(self, size, align)
    }

    method &+self.grow(buffer: &+RawBuffer, new_size: usize): void {
        grow(self, buffer, new_size)
        return
    }
}

instance TryAllocator {
    method &+self.try_alloc(size: usize, align: usize): RawBuffer! {
        return try_alloc(self, size, align)?
    }

    method &+self.try_grow(buffer: &+RawBuffer, new_size: usize): void! {
        try_grow(self, buffer, new_size)?
        return
    }

}
