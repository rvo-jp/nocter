use std/vec.Vec

struct Point {
    x: i32
    y: i32
}

instance Point {
    operator (&self == other: &Self): bool {
        return self.x == other.x && self.y == other.y
    }
}

func main(): i32! {
    let first = Point { x: 3, y: 4 }
    let same = Point { x: 3, y: 4 }
    let other = Point { x: 5, y: 6 }
    if first != same || first == other {
        return 1
    }
    let points = Vec [move first, move other]
    if points[0] != same {
        return 1
    }
    var values = Vec [1, 2, 3]
    values[1] = 4
    if values[0] + values[1] + values[2] != 8 {
        return 1
    }
    io.print("equality found the point\n")?
    return 0
}
struct Buffer {

instance Buffer {
    operator (&self[index: usize]): &i32 {
        return &self.values[index]
    }

    operator (&+self[index: usize]): &+i32 {
        return &+self.values[index]
    }
}

func at<C>(container: &C, index: usize): i32 where (&C[usize]): &i32 {
    return container[index]
}

func main(): i32 {
    var buffer = Buffer { values: [3, 4, 5] }
    buffer[1] = 9
    if at(&buffer, 1) != 9 {
        return 1
    }
    return 0
}

use /internal/mem
use /internal/ptr as internal_ptr

use /internal/safety
use /mem as mem_module
use /ptr

func try_reserve<T>(values: &+Vec<T>, additional: usize): usize! {
    if additional == 0 {
        return values.len
    }

    let values_len: usize = values.len
    let required_capacity: usize = mem.checked_add(values_len, additional) otherwise {
        return mem.capacity_overflow()
    }
    let values_capacity: usize = values.capacity
    if values_capacity >= values_len {
        let spare_capacity: usize = values_capacity - values_len
        if additional <= spare_capacity {
            return required_capacity
        }
    }

    let element_size: usize = internal_ptr.pointee_size(values.ptr)
    if element_size == 0 {
        values.capacity = mem.growth_capacity(values_capacity, required_capacity)
        return required_capacity
    }
    let next_capacity = mem.growth_capacity(values_capacity, required_capacity)
    let byte_capacity: usize = mem.checked_multiply(next_capacity, element_size) otherwise {
        let required_bytes: usize = mem.checked_multiply(required_capacity, element_size) otherwise {
            return mem.capacity_overflow()
        }
        mem_module.try_grow_owned(&+values.storage, required_bytes)?
        values.ptr = internal_ptr.from_addr(ptr.addr(values.storage.bytes().ptr()))
        values.capacity = required_capacity
        return required_capacity
    }
    mem_module.try_grow_owned(&+values.storage, byte_capacity)?
    values.ptr = internal_ptr.from_addr(ptr.addr(values.storage.bytes().ptr()))
    values.capacity = next_capacity
    return required_capacity
}

func reserve<T>(values: &+Vec<T>, additional: usize): void {
    let _ = try_reserve(values, additional) catch allocation_error {
        return mem.allocation_abort()
    }
    return
}

func clear<T>(values: &+Vec<T>): void {
    let element_size: usize = internal_ptr.pointee_size(values.ptr)
    while values.len != 0 {
        let next_len: usize = values.len - 1
        values.len = next_len
        internal_ptr.drop_value_at_ptr(values.ptr, next_len * element_size)
    }
    return
}

func truncate<T>(values: &+Vec<T>, requested_len: usize): void {
    while values.len > requested_len {
        let removed: T = pop(values) otherwise { return }
    }
    return
}

func pop<T>(values: &+Vec<T>): T? {
    if values.len == 0 {
        return none
    }
    let next_len: usize = values.len - 1
    values.len = next_len
    let element_size: usize = internal_ptr.pointee_size(values.ptr)
    return internal_ptr.take_value_at_ptr(values.ptr, next_len * element_size)
}

func try_push<T>(values: &+Vec<T>, value: T): void! {
    let old_len: usize = values.len
    let next_len: usize = try_reserve(values, 1)?
    let element_size: usize = internal_ptr.pointee_size(values.ptr)
    let byte_offset: usize = old_len * element_size
    internal_ptr.store_value_to_ptr(values.ptr, byte_offset, move value)
    values.len = next_len
    return
}

func push<T>(values: &+Vec<T>, value: T): void {
    try_push(values, move value) catch allocation_error {
        return mem.allocation_abort()
    }
    return
}

func try_insert_in_bounds<T>(values: &+Vec<T>, index: usize, value: T): void! {
    let old_len: usize = values.len

    // No fallible operation is permitted after this point. The initialized
    // prefix remains unchanged if capacity growth fails.
    let next_len: usize = try_reserve(values, 1)?
    let element_size: usize = internal_ptr.pointee_size(values.ptr)
    var hole: usize = old_len
    while hole > index {
        let source_index: usize = hole - 1
        let shifted: T = internal_ptr.take_value_at_ptr(values.ptr, source_index * element_size)
        internal_ptr.store_value_to_ptr(values.ptr, hole * element_size, move shifted)
        hole = source_index
    }
    internal_ptr.store_value_to_ptr(values.ptr, index * element_size, move value)
    values.len = next_len
    return
}

func try_insert<T>(values: &+Vec<T>, index: usize, value: T): void! {
    if index > values.len {
        return index_out_of_bounds()
    }
    try_insert_in_bounds(values, index, move value)?
    return
}

func insert<T>(values: &+Vec<T>, index: usize, value: T): void {
    if index > values.len {
        return safety.bounds_abort()
    }
    try_insert_in_bounds(values, index, move value) catch allocation_error {
        return mem.allocation_abort()
    }
    return
}

func remove<T>(values: &+Vec<T>, index: usize): T? {
    let old_len: usize = values.len
    if index >= old_len {
        return none
    }

    let element_size: usize = internal_ptr.pointee_size(values.ptr)
    let removed: T = internal_ptr.take_value_at_ptr(values.ptr, index * element_size)
    var hole: usize = index
    while hole + 1 < old_len {
        let source_index: usize = hole + 1
        let shifted: T = internal_ptr.take_value_at_ptr(values.ptr, source_index * element_size)
        internal_ptr.store_value_to_ptr(values.ptr, hole * element_size, move shifted)
        hole = source_index
    }
    values.len = old_len - 1
    return move removed
}

func swap_remove<T>(values: &+Vec<T>, index: usize): T? {
    let old_len: usize = values.len
    if index >= old_len {
        return none
    }

    let element_size: usize = internal_ptr.pointee_size(values.ptr)
    let last_index: usize = old_len - 1
    let removed: T = internal_ptr.take_value_at_ptr(values.ptr, index * element_size)
    if index != last_index {
        let last: T = internal_ptr.take_value_at_ptr(values.ptr, last_index * element_size)
        internal_ptr.store_value_to_ptr(values.ptr, index * element_size, move last)
    }
    values.len = last_index
    return move removed
}

instance Vec<T> {
    method &+self.reserve(additional: usize): void {
        reserve(self, additional)
        return
    }

    method &+self.try_reserve(additional: usize): void! {
        let _ = try_reserve(self, additional)?
        return
    }

    method &+self.clear(): void {
        clear(self)
        return
    }

    method &+self.truncate(requested_len: usize): void {
        truncate(self, requested_len)
        return
    }

    method &+self.retain<F>(predicate: F): void where F: &+func(&T): bool {
        var predicate_fn = move predicate
        let element_size: usize = internal_ptr.pointee_size(self.ptr)
        let original_len: usize = self.len
        var read_index: usize = 0
        var write_index: usize = 0
        while read_index < original_len {
            let current: &T = &internal_ptr.slice_from_raw_parts_value(self.ptr, original_len)[read_index]
            if predicate_fn(current) {
                if write_index != read_index {
                    let retained: T = internal_ptr.take_value_at_ptr(self.ptr, read_index * element_size)
                    internal_ptr.store_value_to_ptr(self.ptr, write_index * element_size, move retained)
                }
                write_index += 1
            } else {
                internal_ptr.drop_value_at_ptr(self.ptr, read_index * element_size)
            }
            read_index += 1
        }
        self.len = write_index
        return
    }

    method &+self.push(value: T): void {
        push(self, move value)
        return
    }

    method &+self.try_push(value: T): void! {
        try_push(self, move value)?
        return
    }

    method &+self.insert(index: usize, value: T): void {
        insert(self, index, move value)
        return
    }

    method &+self.try_insert(index: usize, value: T): void! {
        try_insert(self, index, move value)?
        return
    }

    method &+self.remove(index: usize): T? {
        return remove(self, index)?
    }

    method &+self.swap_remove(index: usize): T? {
        return swap_remove(self, index)?
    }

    method &+self.pop(): T? {
        return pop(self)?
    }
}

drop Vec<T>(&+self) {
    clear(self)
    self.capacity = 0
    return
}
