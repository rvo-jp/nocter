    }

use std/vec.Vec

struct Point {
    x: i32
    y: i32
}

instance Point {
    operator (&self == other: &Self): bool {
        return self.x == other.x && self.y == other.y
    }
}

func main(): i32! {
    let first = Point { x: 3, y: 4 }
    let same = Point { x: 3, y: 4 }
    let other = Point { x: 5, y: 6 }
    if first != same || first == other {
        return 1
    }
    let points = Vec [move first, move other]
    if points[0] != same {
        return 1
    }
    var values = Vec [1, 2, 3]
    values[1] = 4
    if values[0] + values[1] + values[2] != 8 {
        return 1
    }
    io.print("equality found the point\n")?
    return 0
}
struct Buffer {
    values: [i32; 3]
}

instance Buffer {
    operator (&self[index: usize]): &i32 {
        return &self.values[index]
    }

    operator (&+self[index: usize]): &+i32 {
        return &+self.values[index]
    }
}

func at<C>(container: &C, index: usize): i32 where (&C[usize]): &i32 {
    return container[index]
}

func main(): i32 {
    var buffer = Buffer { values: [3, 4, 5] }
    buffer[1] = 9
    if at(&buffer, 1) != 9 {
        return 1
    }
    return 0
}

//! Darwin filesystem operations and native metadata decoding.

see ./index.nct
see ./filesystem.nct

use /internal/io
use /internal/os.OSErrorKind
use /internal/os/darwin.SyscallResult
use /internal/os/darwin




use /internal/path as internal_path
use /mem.RawBuffer
use /mem
use /ptr
use /internal/ptr as internal_ptr

#target: "arm64-darwin"
struct DarwinStatQuery {
    result: SyscallResult
    storage: RawBuffer
}

#target: "arm64-darwin"
func stat_path(path: &str): DarwinStatQuery! {
    let argument = internal_path.target_path_argument(path)?
    var allocator = mem.page_try_allocator()
    let storage = allocator.try_alloc(darwin.STAT_BUFFER_SIZE, 8)?
    var result = darwin.syscall2(
        darwin.SYS_STAT64,
        internal_path.target_path_address(&argument),
        ptr.addr(storage.bytes().ptr()),
    )
    while result.errno == darwin.ERRNO_INTERRUPTED {
        result = darwin.syscall2(
            darwin.SYS_STAT64,
            internal_path.target_path_address(&argument),
            ptr.addr(storage.bytes().ptr()),
        )
    }
    return DarwinStatQuery { result: result, storage: move storage }
}

#target: "arm64-darwin"
func metadata_target(path: &str): Metadata! {
    let query = stat_path(path)?
    if query.result.errno != 0 {
        return io.io_error_from_os(darwin.error_from_errno(darwin.errno(query.result.errno)))
    }

    let address: usize = ptr.addr(query.storage.bytes().ptr())
    let mode_pointer: *u16 = internal_ptr.from_addr(address)
    let size_pointer: *u64 = internal_ptr.from_addr(address)
    let mode: u16 = internal_ptr.take_value_at_ptr(mode_pointer, darwin.STAT_MODE_OFFSET)
    var file_type = FileType.other
    if mode >= darwin.STAT_MODE_REGULAR && mode < darwin.STAT_MODE_REGULAR + 4096 {
        file_type = FileType.regular
    } else if mode >= darwin.STAT_MODE_DIRECTORY && mode < darwin.STAT_MODE_DIRECTORY + 4096 {
        file_type = FileType.directory
    }
    let len: u64 = internal_ptr.take_value_at_ptr(size_pointer, darwin.STAT_SIZE_OFFSET)
    return Metadata { file_type: file_type, len: len }
}

#target: "arm64-darwin"
func exists_target(path: &str): bool! {
    let query = stat_path(path)?
    if query.result.errno == 0 {
        return true
    }
    let os_error = darwin.error_from_errno(darwin.errno(query.result.errno))
    if os_error.kind is OSErrorKind.not_found {
        return false
    }
    if os_error.kind is OSErrorKind.not_directory {
        return false
    }
    return io.io_error_from_os(os_error)
}

#target: "arm64-darwin"
func remove_file_target(path: &str): void! {
    let argument = internal_path.target_path_argument(path)?
    let result = darwin.syscall1(darwin.SYS_UNLINK, internal_path.target_path_address(&argument))
    if result.errno != 0 {
        return io.io_error_from_os(darwin.error_from_errno(darwin.errno(result.errno)))
    }
    return
}

#target: "arm64-darwin"
func rename_target(source: &str, destination: &str): void! {
    let source_argument = internal_path.target_path_argument(source)?
    let destination_argument = internal_path.target_path_argument(destination)?
    let result = darwin.syscall2(
        darwin.SYS_RENAME,
        internal_path.target_path_address(&source_argument),
        internal_path.target_path_address(&destination_argument),
    )
    if result.errno != 0 {
        return io.io_error_from_os(darwin.error_from_errno(darwin.errno(result.errno)))
    }
    return
}
