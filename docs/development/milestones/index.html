use std/fmt.Format

use std/io

struct Point {
    x: i32
    y: i32
}

instance Point {
    impl Format
    method &self.format_into(output: &+String): void {
        output.push_str("(")
        self.x.format_into(output)
        output.push_str(", ")
        self.y.format_into(output)
        output.push_str(")")
        return
    }
}

func main(): i32! {
    let point = Point { x: 3, y: 4 }
    let message = "point = ${point}\n"
    io.print(&message)?
    return 0
}

use std/vec.Vec

struct Point {
    x: i32
    y: i32
}

instance Point {
    operator (&self == other: &Self): bool {
        return self.x == other.x && self.y == other.y
    }
}

func main(): i32! {
    let first = Point { x: 3, y: 4 }
    let same = Point { x: 3, y: 4 }
    let other = Point { x: 5, y: 6 }
    if first != same || first == other {
        return 1
    }
    let points = Vec [move first, move other]
    if points[0] != same {
        return 1
    }
    var values = Vec [1, 2, 3]
    values[1] = 4
    if values[0] + values[1] + values[2] != 8 {
        return 1
    }
    io.print("equality found the point\n")?
    return 0
}
struct Buffer {
    values: [i32; 3]
}

instance Buffer {
    operator (&self[index: usize]): &i32 {
        return &self.values[index]
    }

    operator (&+self[index: usize]): &+i32 {
        return &+self.values[index]
    }
}

func at<C>(container: &C, index: usize): i32 where (&C[usize]): &i32 {
    return container[index]
}

func main(): i32 {
    var buffer = Buffer { values: [3, 4, 5] }
    buffer[1] = 9
    if at(&buffer, 1) != 9 {
        return 1
    }
    return 0
}

/development/milestones/README.md

Development Milestones

This directory contains the active development milestone, completed milestones, and historical milestone records. Public language behavior belongs in spec/; published qualification belongs in development/releases/.

Active Milestone

The v0.24.0 milestone makes imported value ownership explicit through module-qualified use sites, reserves selected imports for type names, and requires UPPER_SNAKE_CASE constants. Implementation, source migration, qualification, and final review are complete. Its release preparation owns candidate identity, independent source qualification, deterministic packaging, and installed-home evidence. The local candidate is qualified; publication has not been authorized.

Completed Unpublished Milestone

The v0.23.0 milestone completed type-owned decimal parsing and owned text conversion for every built-in integer. Its implementation and full design review are complete; v0.24.0 superseded it before separate release preparation began.

Latest Published Milestones

The v0.22.0 milestone adds an owning JSON DOM, exact decimal tokens, strict parsing, compact generation, practical application integration, and standard-library stabilization. Phases 0 through 5 and release qualification are complete. The release is published and externally audited in the v0.22.0 publication record.

The v0.21.0 milestone returns to practical standard-library work with representation-neutral Map and Set contracts, a private hash-table implementation, and the language and target prerequisites needed to implement them in ordinary Nocter source. Phase 0 through Phase 5 and release qualification are complete; its release preparation owns the frozen candidate identity and qualification evidence. The release is published and externally audited in the v0.21.0 publication record.

Earlier Milestones

The v0.20.0 milestone completed interface prerequisites, incremental semantic computation, unified CLI/LSP query entry, and dependency-local exact selections. It remains an unpublished compiler-foundation record rather than the active work plan.

The v0.19.0 milestone and its release preparation are complete, and v0.19.0 is published. The v0.18.0 milestone and its release preparation are complete, and v0.18.0 is published. The v0.17.0 milestone and its release preparation are complete, and v0.17.0 is published. The v0.16.0 milestone and its release preparation are complete, and v0.16.0 is published. The v0.15.0 milestone and its release preparation are complete, and v0.15.0 is published. The v0.14.0 milestone, its implementation qualification, and the final design review are complete. Earlier v0.14.0 internal refactoring was superseded before publication by a specification-first compiler rewrite. The release preparation is complete.

Every milestone before v0.22.0 records past work only. Historical records are not normative language sources and must not be used to reconstruct behavior missing from the current specification.
