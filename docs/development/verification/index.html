use std/fmt.Format
use std/io

struct Point {
    x: i32
    y: i32
}

instance Point {
    impl Format
    method &self.try_format_into(output: &+String): void! {
        output.try_push_str("(")?
        self.x.try_format_into(output)?
        output.try_push_str(", ")?
        self.y.try_format_into(output)?
        output.try_push_str(")")?
        return
    }
}

func main(): i32! {
    let point = Point { x: 3, y: 4 }
    let message = "point = ${point}\n"
    io.print(&message)?
    return 0
}
use std/io
use std/vec.Vec

struct Point {
    x: i32
    y: i32
}

instance Point {
    operator (&self == other: &Self): bool {
        return self.x == other.x && self.y == other.y
    }
}

func main(): i32! {
    let first = Point { x: 3, y: 4 }
    let same = Point { x: 3, y: 4 }
    let other = Point { x: 5, y: 6 }
    if first != same || first == other {
        return 1
    }
    let points = Vec [move first, move other]
    if points[0] != same {
        return 1
    }
    var values = Vec [1, 2, 3]
    values[1] = 4
    if values[0] + values[1] + values[2] != 8 {
        return 1
    }
    io.print("equality found the point\n")?
    return 0
}
struct Buffer {
    values: [i32; 3]
}

instance Buffer {
    operator (&self[index: usize]): &i32 {
        return &self.values[index]
    }

    operator (&+self[index: usize]): &+i32 {
        return &+self.values[index]
    }
}

func at<C>(container: &C, index: usize): i32 where (&C[usize]): &i32 {
    return container[index]
}

func main(): i32 {
    var buffer = Buffer { values: [3, 4, 5] }
    buffer[1] = 9
    if at(&buffer, 1) != 9 {
        return 1
    }
    return 0
}
/development/verification/README.md

Development Verification

Responsibility

This directory owns repository-development verification entry points whose intermediate artifacts must not become persistent repository caches. Release assembly and artifact qualification remain under development/packaging/.

Compiler Verification

Run the complete compiler gate from any directory:

development/verification/verify-compiler.sh

The script creates one target under /tmp, shares it across formatting, warnings-denied Clippy, workspace tests, feature checking, and Rust documentation, then removes it on exit. A complete gate therefore cannot add another Cargo hash generation to development/compiler/target/.

Use the workspace target only for focused inner-loop commands. It is a disposable cache; reclaim it without affecting source or release artifacts with:

cargo clean --manifest-path development/compiler/Cargo.toml

The script intentionally does not clean the workspace target. Verification must not destroy a developer's active inner-loop cache as a side effect.
