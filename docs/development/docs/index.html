use std/io.print

func greet(user: User): void! {
    let name = user.display.name() otherwise { "guest" }

    let tone = match user.presence {
        Presence.online { "hello" }
        Presence.away { "welcome back" }
    }

    print("${tone}, ${name}\n")?
}
interface Writer {
    pub method &+self.write(bytes: &[u8]): void!
}

func save_log(output: &+Writer, events: &[Event]): void! {
    for event in events {
        output.write(event.message().bytes())?
        output.write("\n".bytes())?
    }
}
pub struct Session {
    id: SessionId
    user: User
    expires_at: Time
}

pub func Session.start(user: User, clock: &Clock): Session {
    return Session {
        id: SessionId.new(),
        user: move user,
        expires_at: clock.now().plus_minutes(30),
    }
}
struct Profile {
    ...UserSummary
    ...ActivityStats
    visits: u32
}

func profile(user: &User, stats: ActivityStats): Profile {
    return Profile {
        ...user.summary(),
        ...move stats,
        visits: 0,
    }
}
literal NonEmptyList<T> [first: T, ...rest: [T]]: Self {
    let list = Self.with_first(move first)

    for item in rest {
        list.push(move item)
    }

    return list
}

Nocter Development Documents

このディレクトリは実装者向けの設計と進行条件だけを扱う。言語の公開仕様は spec が唯一の規範であり、ここへ複製しない。

現行の開発マイルストーンは v0.2.0 のみである。本文中で v0 を リリース名や作業範囲の略称として使わない。

Documents

  • v0.2.0 Development Contract: 終了条件、非目標、実装順序、 リリース判定。進捗判断の唯一の入口。
  • Architecture: コンパイラのフェーズ責務と境界。
  • Allocator and Ownership: メモリ確保、所有値、部分初期化、 String と Vec<T> を成立させる共通基盤。
  • Standard Library: 配布標準ライブラリの現状と v0.2.0 の 実行時受入条件。
  • LSP: コンパイラ解析を再利用する LSP の設計と v0.2.0 の受入条件。
  • Maintenance: 更新責務、検証、コミット規約。
  • TODO: 次の作業に必要な短期引き継ぎ。
