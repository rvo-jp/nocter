use std/io.print

func greet(user: User): void! {
    let name = user.display.name() otherwise { "guest" }

    let tone = match user.presence {
        Presence.online { "hello" }
        Presence.away { "welcome back" }
    }

    print("${tone}, ${name}\n")?
}
interface Writer {
    pub method &+self.write(bytes: &[u8]): void!
}

func save_log(output: &+Writer, events: &[Event]): void! {
    for event in events {
        output.write(event.message().bytes())?
        output.write("\n".bytes())?
    }
}
pub struct Session {
    id: SessionId
    user: User
    expires_at: Time
}

pub func Session.start(user: User, clock: &Clock): Session {
    return Session {
        id: SessionId.new(),
        user: move user,
        expires_at: clock.now().plus_minutes(30),
    }
}
struct Profile {
    ...UserSummary
    ...ActivityStats
    visits: u32
}

func profile(user: &User, stats: ActivityStats): Profile {
    return Profile {
        ...user.summary(),
        ...move stats,
        visits: 0,
    }
}
literal NonEmptyList<T> [first: T, ...rest: [T]]: Self {
    let list = Self.with_first(move first)

    for item in rest {
        list.push(move item)
    }

    return list
}

Nocter Development Documents

This directory contains implementation design and completion criteria. The public language specification is the sole authority for language semantics; do not duplicate those rules here.

The released baseline is v0.2.0. v0.3.0 Phase 0 and Phase 1: Typed Literal Core are complete on develop; Phase 2: Explicit Iteration and Collection Access is active. Do not use v0 as shorthand for a release name or work scope.

Documents

  • v0.3.0 Development Contract: completed Phase 0 and Phase 1 records plus the active Phase 2 gate; the single entry point for current milestone status
  • v0.2.0 Release Record: immutable completion criteria for the released baseline
  • Architecture: compiler phase responsibilities and boundaries
  • Region, Provenance, and Allocation Context: shared storage-origin, allocation-effect, lexical-region, and lowering design
  • Typed Literal Core: literal shapes, definitions, element packs, context selection, and lowering boundaries
  • Explicit Iteration and Collection Access: readonly and owned iterator invariants, optional access, shifting, cleanup, and LSP boundaries
  • Allocator and Ownership: the shared allocation, ownership, partial initialization, String, and Vec<T> foundation
  • Standard Library: released runtime baseline and completed Phase 0 allocator-policy integration
  • LSP: released compiler-backed capabilities and completed Phase 0 region integration
  • Maintenance: update ownership, verification, and commit policy
  • TODO: internal short-term handoff state
