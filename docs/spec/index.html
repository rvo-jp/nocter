use std/fmt.Format
use std/io

struct Point {
    x: i32
    y: i32
}

instance Point {
    impl Format
    method &self.try_format_into(output: &+String): void! {
        output.try_push_str("(")?
        self.x.try_format_into(output)?
        output.try_push_str(", ")?
        self.y.try_format_into(output)?
        output.try_push_str(")")?
        return
    }
}

blocking func main(): i32! {
    let point = Point { x: 3, y: 4 }
    let message = "point = ${point}\n"
    io.print(&message)?
    return 0
}
use std/io
use std/vec.Vec

struct Point {
    x: i32
    y: i32
}

instance Point {
    operator (&self == other: &Self): bool {
        return self.x == other.x && self.y == other.y
    }
}

blocking func main(): i32! {
    let first = Point { x: 3, y: 4 }
    let same = Point { x: 3, y: 4 }
    let other = Point { x: 5, y: 6 }
    if first != same || first == other {
        return 1
    }
    let points = Vec [move first, move other]
    if points[0] != same {
        return 1
    }
    var values = Vec [1, 2, 3]
    values[1] = 4
    if values[0] + values[1] + values[2] != 8 {
        return 1
    }
    io.print("equality found the point\n")?
    return 0
}
struct Buffer {
    values: [i32; 3]
}

instance Buffer {
    operator (&self[index: usize]): &i32 {
        return &self.values[index]
    }

    operator (&+self[index: usize]): &+i32 {
        return &+self.values[index]
    }
}

func at<C>(container: &C, index: usize): i32 where (&C[usize]): &i32 {
    return container[index]
}

func main(): i32 {
    var buffer = Buffer { values: [3, 4, 5] }
    buffer[1] = 9
    if at(&buffer, 1) != 9 {
        return 1
    }
    return 0
}

Nocter Specification

This directory is the sole normative source for the current Nocter language, supported platform boundary, command-line interface, and editor behavior. The checked standard-library source and its documentation are owned separately by the Standard Library. The catalogs below recommend a reading order; generated navigation follows the published directory structure and does not use those links as a page registry. File names describe responsibilities and never encode release history.

Nocter is a statically typed, value-centered systems language designed for self-contained native executables, explicit contracts, and a small installation surface. The language favors one canonical source style and makes machine-readable diagnostics and editor semantics part of the public contract.

Start Here

  • Design Principles explains the criteria used to resolve design choices.
  • Language Overview introduces the language as a coherent whole.
  • Language defines source syntax and program semantics.
  • Standard Library defines portable public modules and APIs next to their checked declarations.
  • Platform defines ABI, target, primitive, and distribution boundaries.
  • Tooling defines diagnostics, formatting, CLI, testing, and editor behavior.
  • Runnable Examples contains complete programs checked against released behavior.

Contract Status

The working tree specifies the in-development v0.45.0 contract. The published v0.44.0 contract remains available from its repository tag. The release index owns publication status, downloads, and version summaries; repository tags preserve the exact specification for earlier releases.

A chapter states current behavior unless a section is explicitly labeled Future Direction or Non-goal. Milestones, implementation plans, reviews, and release qualification evidence are non-normative and live under development/history/.

Editing Policy

  • Put each public rule in exactly one owning chapter and link to it instead of restating it.
  • Keep language semantics, library APIs, platform boundaries, and tooling contracts in their respective directories.
  • Keep examples close to the rule they explain, while complete runnable packages remain under examples/.
  • Do not encode chronology or implementation phases in chapter names or current-contract prose.
  • Mark proposals under an explicit Future Direction heading.
  • Keep compiler mechanisms and repository workflows under development/.
